cmake_minimum_required(VERSION 3.24)
project(pygo_cuda LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
file(GLOB CUDA_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cu" "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_library(pygo_cuda STATIC ${CUDA_SOURCES})
target_include_directories(pygo_cuda PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../include")
set_target_properties(pygo_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION ON)
target_compile_options(pygo_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
