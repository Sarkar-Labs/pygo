cmake_minimum_required(VERSION 3.20)
project(pygo VERSION 0.1.0 LANGUAGES C CXX)

set(PYGO_ENABLE_CUDA "AUTO" CACHE STRING "CUDA support: AUTO, ON, or OFF")
set_property(CACHE PYGO_ENABLE_CUDA PROPERTY STRINGS AUTO ON OFF)
option(PYGO_ENABLE_XLA "Enable XLA/TPU integration hooks (no TPU runtime is bundled)" OFF)
option(PYGO_BUILD_LEGACY "Compile legacy experimental native modules" OFF)
option(PYGO_BUILD_TESTS "Build native tests" ON)
option(PYGO_BUILD_SHARED "Build the production shared library" ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
find_package(Python3 COMPONENTS Interpreter QUIET)

file(GLOB_RECURSE PYGO_TENSOR_CPP CONFIGURE_DEPENDS cpp/tensor/*.cpp)
file(GLOB_RECURSE PYGO_ACTIVATION_CPP CONFIGURE_DEPENDS cpp/activations/*.cpp)
file(GLOB_RECURSE PYGO_INITIALIZER_CPP CONFIGURE_DEPENDS cpp/initializers/*.cpp)
file(GLOB_RECURSE PYGO_METRIC_CPP CONFIGURE_DEPENDS cpp/metrics/*.cpp)
set(PYGO_CPP ${PYGO_TENSOR_CPP} ${PYGO_ACTIVATION_CPP} ${PYGO_INITIALIZER_CPP} ${PYGO_METRIC_CPP} cpp/kernels/reference_kernels.cpp)


if(PYGO_BUILD_SHARED)
  add_library(pygo_native SHARED ${PYGO_CPP})
else()
  add_library(pygo_native STATIC ${PYGO_CPP})
endif()
set_target_properties(pygo_native PROPERTIES OUTPUT_NAME pygo)
target_include_directories(pygo_native PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp/include> $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/cpp> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(pygo_native PUBLIC cxx_std_17)
target_compile_definitions(pygo_native PUBLIC PYGO_PRODUCTION_BUILD=1)

add_library(pygo_c_api_obj OBJECT cpp/bridge/c_api.cpp)
target_include_directories(pygo_c_api_obj PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include ${CMAKE_CURRENT_SOURCE_DIR}/cpp ${CMAKE_CURRENT_SOURCE_DIR}/cpp/bridge)
target_compile_features(pygo_c_api_obj PRIVATE cxx_std_17)
target_link_libraries(pygo_native PRIVATE pygo_c_api_obj)

if(PYGO_BUILD_SHARED)
  set_target_properties(pygo_native PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()
include(CheckLanguage)
check_language(CUDA)
set(PYGO_CUDA_ENABLED OFF)
if(PYGO_ENABLE_CUDA STREQUAL "ON")
  if(NOT CMAKE_CUDA_COMPILER)
    message(FATAL_ERROR "PYGO_ENABLE_CUDA=ON was requested, but no CUDA toolkit/nvcc was found. Install CUDA or use PYGO_ENABLE_CUDA=AUTO/OFF.")
  endif()
  enable_language(CUDA)
  set(PYGO_CUDA_ENABLED ON)
elseif(PYGO_ENABLE_CUDA STREQUAL "AUTO")
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(PYGO_CUDA_ENABLED ON)
    message(STATUS "PyGo: CUDA toolkit detected; enabling CUDA backend")
  else()
    message(STATUS "PyGo: CUDA toolkit not detected; building CPU-only backend")
  endif()
elseif(NOT PYGO_ENABLE_CUDA STREQUAL "OFF")
  message(FATAL_ERROR "PYGO_ENABLE_CUDA must be AUTO, ON, or OFF")
endif()
if(PYGO_CUDA_ENABLED)
  file(GLOB_RECURSE PYGO_CUDA CONFIGURE_DEPENDS cpp/cuda/*.cu)
  target_sources(pygo_native PRIVATE ${PYGO_CUDA})
  set_target_properties(pygo_native PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  target_compile_definitions(pygo_native PUBLIC PYGO_CUDA_ENABLED=1)
  set_target_properties(pygo_native PROPERTIES CUDA_SEPARABLE_COMPILATION ON POSITION_INDEPENDENT_CODE ON)
  if(Python3_Interpreter_FOUND)
    add_custom_target(pygo_cuda_validate
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/cuda/validate_cuda_sources.py --nvcc ${CMAKE_CUDA_COMPILER}
      WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
      USES_TERMINAL
      COMMENT "Compiling every PyGo CUDA translation unit with nvcc")
  endif()
else()
  target_compile_definitions(pygo_native PUBLIC PYGO_CUDA_ENABLED=0)
endif()
if(PYGO_ENABLE_XLA)
  message(WARNING "PyGo XLA/TPU integration hooks are enabled at the API level only; no TPU runtime or PJRT backend is bundled in this release.")
  target_compile_definitions(pygo_native PUBLIC PYGO_XLA_ENABLED=1)
else()
  target_compile_definitions(pygo_native PUBLIC PYGO_XLA_ENABLED=0)
endif()

add_library(pygo::native ALIAS pygo_native)

if(PYGO_BUILD_LEGACY)
  message(WARNING "PYGO_BUILD_LEGACY is experimental and excluded from the production ABI.")
  file(GLOB_RECURSE PYGO_LEGACY_CPP CONFIGURE_DEPENDS cpp/core/*.cpp cpp/autograd/*.cpp cpp/nn/*.cpp cpp/optim/*.cpp cpp/loss/*.cpp cpp/distributed/*.cpp cpp/serialization/*.cpp cpp/serving/*.cpp)
  add_library(pygo_legacy OBJECT ${PYGO_LEGACY_CPP})
  target_include_directories(pygo_legacy PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cpp/include ${CMAKE_CURRENT_SOURCE_DIR}/cpp)
  target_compile_features(pygo_legacy PRIVATE cxx_std_17)
endif()

include(CTest)
enable_testing()
if(PYGO_BUILD_TESTS)
  add_executable(pygo_native_smoke cpp/tests/native_smoke.cpp)
  target_link_libraries(pygo_native_smoke PRIVATE pygo_native)
  add_test(NAME pygo_native_smoke COMMAND pygo_native_smoke)
  add_executable(pygo_c_api_smoke cpp/tests/c_api_smoke.cpp)
  target_link_libraries(pygo_c_api_smoke PRIVATE pygo_native)
  target_include_directories(pygo_c_api_smoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cpp/bridge)
  add_test(NAME pygo_c_api_smoke COMMAND pygo_c_api_smoke)
endif()

include(CMakePackageConfigHelpers)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/PyGoConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
configure_package_config_file(cmake/PyGoConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/PyGoConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PyGo)

install(TARGETS pygo_native EXPORT PyGoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY cpp/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES cpp/bridge/c_api.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/pygo)
install(EXPORT PyGoTargets NAMESPACE pygo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PyGo)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/PyGoConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/PyGoConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PyGo)
